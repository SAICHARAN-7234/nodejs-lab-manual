Session Tracking using Cookies and HTTP Session

Demonstrates tracking user visits across requests using browser Cookies and server-side HTTP Session attributes.

Live Session Tracking Demo

Visit #1
Cookie Name: student Cookie Value: Ravi
Session ID (JSESSIONID): JSESS_8F3A29B10C4
Cookie Max-Age: 3600 seconds (1 hour)

Servlet Implementation (SessionServlet.java)

@WebServlet("/session")
public class SessionServlet extends HttpServlet {
 protected void doGet(HttpServletRequest req, HttpServletResponse res) throws IOException {
  res.setContentType("text/html");
  HttpSession session = req.getSession();
  Integer count = (Integer)session.getAttribute("count");
  count = (count == null) ? 1 : count + 1;
  session.setAttribute("count", count);

  Cookie cookie = new Cookie("student", "Ravi");
  cookie.setMaxAge(3600);
  res.addCookie(cookie);

  try(PrintWriter out = res.getWriter()){
   out.println("<h2>Session Tracking Demo</h2>");
   out.println("<p>HTTP Session visit count: " + count + "</p>");
   out.println("<p>Cookie 'student' has been sent with value: Ravi</p>");
   out.println("<p>Session ID: " + session.getId() + "</p>");
  }
 }
}
